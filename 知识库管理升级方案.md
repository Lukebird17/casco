## 🔍 详细实现

### API实现示例

#### 1. 获取知识库文件列表

```python
@app.get("/api/knowledge-bases/{kb_id}/files")
async def list_kb_files(kb_id: str):
    """获取指定知识库的文件列表"""
    try:
        from config import get_kb_data_dir
        import glob
        
        data_dir = get_kb_data_dir(kb_id)
        
        if not os.path.exists(data_dir):
            return {
                "success": True,
                "files": [],
                "count": 0
            }
        
        files = []
        for filepath in glob.glob(os.path.join(data_dir, "*")):
            if os.path.isfile(filepath):
                filename = os.path.basename(filepath)
                file_size = os.path.getsize(filepath) / (1024 * 1024)  # MB
                file_ext = os.path.splitext(filename)[1].lower()
                
                files.append({
                    "name": filename,
                    "size": round(file_size, 2),
                    "type": file_ext[1:],  # 去掉点号
                    "path": filepath,
                    "created_at": os.path.getctime(filepath)
                })
        
        # 按创建时间倒序
        files.sort(key=lambda x: x['created_at'], reverse=True)
        
        return {
            "success": True,
            "kb_id": kb_id,
            "files": files,
            "count": len(files)
        }
        
    except Exception as e:
        print(f"❌ 获取文件列表错误: {e}")
        raise HTTPException(status_code=500, detail=str(e))
```

#### 2. 删除知识库文件

```python
@app.delete("/api/knowledge-bases/{kb_id}/files/{filename}")
async def delete_kb_file(kb_id: str, filename: str):
    """删除原文件和向量记录"""
    try:
        from config import get_kb_data_dir, get_kb_vector_dir
        from vector_store import VectorStore
        
        # 1. 删除原文件
        data_dir = get_kb_data_dir(kb_id)
        file_path = os.path.join(data_dir, filename)
        
        if not os.path.exists(file_path):
            raise HTTPException(status_code=404, detail="文件不存在")
        
        os.remove(file_path)
        print(f"✅ 删除原文件: {file_path}")
        
        # 2. 删除向量记录
        vector_dir = get_kb_vector_dir(kb_id)
        vector_store = VectorStore(db_path=vector_dir)
        
        # 根据filename查询并删除相关的向量
        # 注意：这需要vector_store支持按filename删除
        # 可能需要添加新方法: vector_store.delete_by_filename(filename)
        
        try:
            # 假设有这个方法
            deleted_count = vector_store.delete_by_metadata(
                {"filename": filename}
            )
            print(f"✅ 删除向量记录: {deleted_count} 条")
        except Exception as e:
            print(f"⚠️  删除向量记录失败: {e}")
        
        return {
            "success": True,
            "message": f"文件 '{filename}' 已删除",
            "kb_id": kb_id,
            "filename": filename
        }
        
    except HTTPException:
        raise
    except Exception as e:
        print(f"❌ 删除文件错误: {e}")
        import traceback
        traceback.print_exc()
        raise HTTPException(status_code=500, detail=str(e))
```

## ⚠️ 注意事项

1. **vector_store需要支持按metadata删除**
   - 需要在VectorStore类中添加`delete_by_metadata`方法
   - 或者在添加文档时存储filename作为metadata

2. **数据迁移不可逆**
   - 运行迁移脚本前会自动备份
   - 测试后再删除备份

3. **前端状态同步**
   - 切换知识库后需要通知App组件
   - 可能需要使用Context或Redux

4. **文件下载vs在线查看**
   - iframe + `Content-Disposition: inline` 可解决
   - 或使用专门的文档查看库
