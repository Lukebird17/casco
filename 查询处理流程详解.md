```python
def analyze_query_type(query):
    # 检查技术术语
    technical_terms = self.extract_technical_terms(query)
    
    # 检查问题复杂度
    if len(technical_terms) <= 1 and "是什么" in query:
        return "basic"      # 简单问题
    elif "比较" in query or "区别" in query:
        return "intermediate"  # 中等复杂度
    else:
        return "advanced"   # 高复杂度
```

#### 3.3 多查询检索（核心！）

```python
def multi_query_retrieve(query, k):
    # 1. 查询增强 - 生成多个查询变体
    queries = self.enhance_query(query)
    # 原始: "第28页的slab示意图中的Next_free是什么？"
    # 生成:
    #   - "slab示意图 Next_free"
    #   - "第28页 slab分配器"
    #   - "Next_free 指针 slab"
    
    all_results = []
    
    # 2. 对每个查询执行检索
    for q in queries:
        # ⭐ 关键：使用混合检索器（支持多模态）
        if self.use_multimodal and self.hybrid_retriever:
            search_results = self.hybrid_retriever.search(
                query=q,
                top_k=k,
                include_images=True  # 包含图片结果！
            )
            results = search_results['combined']  # 文本 + 图片
        else:
            results = self.vector_store.search(q, top_k=k)  # 仅文本
        
        all_results.extend(results)
    
    # 3. 去重
    return deduplicate(all_results)
```

#### 4.2 图片向量库检索 (image_vector_store.py)

```python
def search_by_text(text, top_k):
    # 1. 使用 CLIP 模型编码文本
    text_embedding = self._encode_text(text)
    # CLIP 可以将文本和图片映射到同一向量空间
    
    # 2. 在图片向量库中搜索
    results = self.collection.query(
        query_embeddings=[text_embedding],
        n_results=top_k
    )
    
    # 3. 格式化结果
    image_results = []
    for i, metadata in enumerate(results['metadatas'][0]):
        image_results.append({
            'type': 'image',
            'image_path': metadata['image_path'],
            'description': results['documents'][0][i],
            'filename': metadata['filename'],
            'page_number': metadata['page_number'],
            'distance': results['distances'][0][i]
        })
    
    return image_results
```

**CLIP 编码原理**：
```
文本: "slab示意图 Next_free"
    ↓
CLIP Text Encoder
    ↓
512维向量: [0.23, -0.15, 0.88, ...]
    ↓
在图片向量空间中搜索相似向量
    ↓
找到语义相似的图片
```

### 5. 结果重排序

```python
def rerank_results(query, results):
    # 1. 提取关键词
    keywords = extract_keywords(query)  # ["slab", "Next_free", "28页"]
    
    # 2. 计算匹配度
    for result in results:
        if result['type'] == 'text':
            content = result['content']
        else:  # image
            content = result['description']
        
        # 计算关键词匹配比例
        matches = sum(1 for kw in keywords if kw in content)
        result['match_ratio'] = matches / len(keywords)
    
    # 3. 按匹配度排序
    results.sort(key=lambda x: x['match_ratio'], reverse=True)
    
    return results
```

### 6. 上下文格式化

```python
def _format_context(results):
    context_parts = []
    
    for i, doc in enumerate(results, 1):
        if doc.get('type') == 'image':
            # 图片结果
            context_parts.append(
                f"参考资料 {i}（图片）：\n"
                f"[图片]\n"
                f"路径: {doc['image_path']}\n"
                f"描述: {doc['description']}\n"
                f"来源：《{doc['filename']}》第 {doc['page_number']} 页\n"
            )
        else:
            # 文本结果
            context_parts.append(
                f"参考资料 {i}：\n"
                f"{doc['content']}\n"
                f"来源：《{doc['filename']}》第 {doc['page_number']} 页\n"
            )
    
    return "\n".join(context_parts)
```

### 7. 生成回答

```python
def generate_response(query, context, chat_history, query_type):
    # 1. 构建系统提示
    system_prompt = self._build_system_prompt(query_type)
    
    # 2. 构建消息列表
    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": f"问题: {query}\n\n{context}"}
    ]
    
    # 3. 调用LLM
    response = self.client.chat.completions.create(
        model=self.model,
        messages=messages,
        temperature=0.3
    )
    
    return response.choices[0].message.content
```

```python
def chat_with_file(message, file, history):
    # 1. 处理文件
    processor = EnhancedOCRProcessor()
    result = processor.process_file(file.name)
    
    # 2. 分块
    text_splitter = TextSplitter(chunk_size=500, chunk_overlap=50)
    chunks = text_splitter.split_text(result['content'])
    
    # 3. 临时向量化
    temp_store = VectorStore(db_path="temp_db", collection_name="temp")
    for i, chunk in enumerate(chunks):
        temp_store.add_documents([{
            'content': chunk,
            'filename': file.name,
            'page_number': i,
            'chunk_id': i,
            'filepath': file.name,
            'filetype': 'pdf'
        }])
    
    # 4. 使用临时库检索
    results = temp_store.search(message, top_k=3)
    context = format_context(results)
    
    # 5. 生成回答
    answer = agent.generate_response(message, context, history)
    
    # 6. 清理临时库
    temp_store.clear()
    
    return answer
```

## 💡 关键技术点

### 1. CLIP 模型的作用

**为什么使用 CLIP？**
- 可以将文本和图片映射到同一向量空间
- 支持跨模态检索（以文搜图、以图搜文）
- 多语言支持（中英文都可以）

**CLIP 编码示例**：
```python
# 文本编码
text = "slab分配器示意图"
text_embedding = clip_model.encode_text(text)
# → [0.23, -0.15, 0.88, ..., 0.42] (512维)

# 图片编码
image = Image.open("slab_diagram.jpg")
image_embedding = clip_model.encode_image(image)
# → [0.21, -0.14, 0.90, ..., 0.40] (512维)

# 计算相似度
similarity = cosine_similarity(text_embedding, image_embedding)
# → 0.85 (高度相似！)
```

---

### 2. 双索引系统

**为什么需要两个向量库？**

```
文本向量库                      图片向量库
- bge-m3 (1024维)              - CLIP (512维)
- 优化文本语义理解              - 优化图片-文本对齐
- 存储文档块                    - 存储图片+描述
```

**优势**：
- 各自使用最适合的模型
- 可以独立优化
- 检索时可以并行处理

---

### 3. 结果合并策略

**交替插入法**：
```python
def _merge_results(text_results, image_results, top_k):
    combined = []
    i, j = 0, 0
    
    while len(combined) < top_k:
        # 交替添加
        if i < len(text_results):
            combined.append(text_results[i])
            i += 1
        if j < len(image_results) and len(combined) < top_k:
            combined.append(image_results[j])
            j += 1
    
    return combined
```

**为什么交替插入？**
- 保持结果多样性
- 避免全是文本或全是图片
- 给用户提供更全面的信息
